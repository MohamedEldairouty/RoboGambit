#

#        See https://docs.ros.org/en/humble/Installation.html
#
#   2. Source the ROS setup before running the GUI:
#        source /opt/ros/$ROS_DISTRO/setup.bash
#
#   3. Make sure your venv was created with --system-site-packages so it
#      can see the system-installed rclpy:
#        python3 -m venv venv --system-site-packages
#
# If rclpy isn't available, ROSRobot prints a warning and the GUI keeps
# running in no-op publish mode — it won't crash.
